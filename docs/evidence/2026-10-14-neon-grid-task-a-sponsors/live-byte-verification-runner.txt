base='https://manaconda33.github.io/manacondas-minigame-mayhem/previews/neon-grid-task-a-001/'
root=Path('/tmp/task-a-preview-dist');expected=json.loads(Path('/tmp/neon-grid-task-a/docs/evidence/2026-10-14-neon-grid-task-a-sponsors/prepared-preview-hashes.json').read_text())

def fetch(path):
 for attempt in range(3):
  try:
   request=urllib.request.Request(base+path+'?verify=task-a-692e942',headers={'Cache-Control':'no-cache','User-Agent':'codex-task-a-byte-verification'})
   with urllib.request.urlopen(request,timeout=45) as response:
    data=response.read();status=response.status
   assert status==200,path
   digest=hashlib.sha256(data).hexdigest();assert digest==expected[path],f'Byte mismatch: {path}'
   return {'path':path,'http':status,'bytes':len(data),'sha256':digest}
  except urllib.error.HTTPError as error:
   if error.code not in [429,500,502,503,504] or attempt==2:raise
   time.sleep(2**attempt)
